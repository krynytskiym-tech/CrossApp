7. ДОДАТКОВІ ЗАВДАННЯ
-------------------------------------------------
    # 1. Join і GroupJoin
    dotnet run --no-build --project src/Cli -- report-movements
    # Join: 5 рядків (SKU-001 Issued -20, SKU-002 Received +10 і +5,
    #        SKU-003 Issued -100, SKU-016 Received +15)

    dotnet run --no-build --project src/Cli -- report-activity
    # GroupJoin: 4 товари з рухом і "Без руху: 14 позицій"

    # 2. Метод-розширення TopBy у двох звітах

    # 3. Синтаксис запитів: результат збігається з report-groups

    dotnet run --no-build --project src/Cli -- report-groups-query > /tmp/g2.txt
